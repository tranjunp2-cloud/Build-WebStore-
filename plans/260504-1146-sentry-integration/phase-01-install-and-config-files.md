## Implementation steps

### 1. Install package

```bash

```

Pin exact version — Sentry has had breaking changes between minors historically.

### 2. Create `sentry.server.config.ts` (root)

```ts
import * as Sentry from '@sentry/nextjs'

const dsn = process.env.SENTRY_DSN
const env = process.env.SENTRY_ENVIRONMENT || process.env.APP_ENV || 'development'

const sampleByEnv = (e: string) =>
  e === 'production' ? 0.1 : e === 'staging' ? 0.5 : 1.0

Sentry.init({
  dsn,
  environment: env,
  tracesSampleRate: sampleByEnv(env),
  integrations: [
    Sentry.postgresIntegration(),
  ],
  sendDefaultPii: false,
  maxBreadcrumbs: 50,
  beforeSend(event) {
    // Drop 4xx response errors to save quota
    const status = (event.contexts?.response as { status_code?: number } | undefined)?.status_code
    if (status && status >= 400 && status < 500) return null

    // Strip body for media upload routes (multipart)
    const url = event.request?.url || ''
    if (/\/api\/media\//.test(url) && event.request) {
      delete event.request.data
    }

    // Strip auth cookie defensively
    if (event.request?.cookies && typeof event.request.cookies === 'object') {
      delete (event.request.cookies as Record<string, unknown>)['payload-token']
    }
    return event
  },
})
```

Note: `Sentry.postgresIntegration()` (Sentry 10.x replaces `PgIntegration` from earlier docs). Confirm exact name from `@sentry/nextjs` 10.50.0 exports — fall back to `Sentry.nodeContextIntegration()` if API moved. Acceptance: file compiles via `npm run typecheck`.

### 3. Create `sentry.client.config.ts` (root)

```ts
import * as Sentry from '@sentry/nextjs'

const dsn = process.env.NEXT_PUBLIC_SENTRY_DSN
const env = process.env.NEXT_PUBLIC_APP_ENV || 'development'

Sentry.init({
  dsn,
  environment: env,
  tracesSampleRate: env === 'production' ? 0.1 : env === 'staging' ? 0.5 : 1.0,
  sendDefaultPii: false,
  beforeSend(event) {
    // Drop expected client noise (extension errors, ResizeObserver loop)
    const msg = event.exception?.values?.[0]?.value || ''
    if (/ResizeObserver loop|Non-Error promise rejection captured/.test(msg)) return null
    return event
  },
})
```

### 5. Create `instrumentation.ts` (root)

```ts
export async function register() {
  if (process.env.NEXT_RUNTIME === 'nodejs') {
    await import('./sentry.server.config')
  }
  if (process.env.NEXT_RUNTIME === 'edge') {
    await import('./sentry.edge.config')
  }
}

export { onRequestError } from '@sentry/nextjs'
```

`onRequestError` re-export is the Next.js 15 hook for capturing request errors automatically.

### 7. Create `src/app/global-error.tsx`

```tsx
'use client'

import * as Sentry from '@sentry/nextjs'
import { useEffect } from 'react'

export default function GlobalError({
  error,
}: {
  error: Error & { digest?: string }
}) {
  useEffect(() => {
    Sentry.captureException(error)
  }, [error])

  return (
    <html>
      <body>
        <h2>Something went wrong</h2>
        <p>Our team has been notified.</p>
      </body>
    </html>
  )
}
```

KISS: no recovery button — Next.js auto-recovers on next nav. Add reset button only if UX requires.

### 8. Create `src/lib/ai/sentry-spans.ts`

```ts
import * as Sentry from '@sentry/nextjs'

/**
 * Wrap an async AI gen step in a Sentry span.
 * No-op if Sentry not initialized (DSN missing locally).
 *
 * Usage:
 *   const result = await withSentrySpan('ai.generateSite', () => generateSite(input))
 */
export async function withSentrySpan<T>(name: string, fn: () => Promise<T>): Promise<T> {
  return Sentry.startSpan({ name, op: 'ai' }, async () => fn())
}
```

Do NOT modify any AI files yet — wiring into `generateSite()` happens in Phase 04 verification (lowest-risk integration point with smoke test). Keeps Phase 01 strictly additive.
